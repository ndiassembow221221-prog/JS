/* ╔══════════════════════════════════════════════════════════════════════╗
//    ║          DOCUMENTATION CSS TRANSFORM – COMPLÈTE (2D & 3D)           ║
//    ╚══════════════════════════════════════════════════════════════════════╝

//   La propriété CSS `transform` modifie l'apparence visuelle d'un élément
//   dans l'espace (2D ou 3D) SANS perturber le flux du document.
//   Les autres éléments ne "voient" pas le déplacement : aucun reflow.

//   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
//   A. SYNTAXE DE BASE
//   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

//     .element {
//       transform: fonction1(...) fonction2(...);
//     }

//   Plusieurs fonctions sont chaînées séparées par un espace.
//   Elles s'appliquent DE DROITE À GAUCHE (ordre mathématique des matrices).

//   Exemple :
//     transform: rotate(45deg) translateX(100px);
//     → d'abord translateX(100px), ENSUITE rotate(45deg)
//     (l'élément se déplace à 100px dans son propre axe X avant d'être pivoté)

//   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
//   B. TRANSFORMS 2D
//   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

//   1. translate(x, y) / translateX(x) / translateY(y)
//      ─────────────────────────────────────────────────
//      Déplace l'élément sur les axes X et/ou Y.
//      Unités : px, %, em, rem, vw, vh…
//      Le % est relatif à la TAILLE DE L'ÉLÉMENT LUI-MÊME (pas du parent).

//      transform: translate(50px, -20px);     /* droite 50, haut 20 *
//      transform: translateX(50%);            /* 50% de sa propre largeur *
//      transform: translateY(-100%);          /* monte de sa propre hauteur *

//      Usage classique : centrage absolu
//        position: absolute; top: 50%; left: 50%;
//        transform: translate(-50%, -50%);    /* recule de sa propre moitié 

//   2. scale(x, y) / scaleX(x) / scaleY(y)
//      ─────────────────────────────────────
//      Redimensionne l'élément. Valeur sans unité.
//        1   = taille originale
//        2   = double
//        0.5 = moitié
//       -1   = miroir (flip)

//      transform: scale(1.5);            /* uniforme : 1.5× dans les 2 axes 
//      transform: scale(2, 0.5);         /* large × 2, hauteur ÷ 2 *
//      transform: scaleX(-1);            /* miroir horizontal *
//      transform: scaleY(-1);            /* miroir vertical *

//   3. rotate(angle)
//      ──────────────
//      Rotation dans le plan (autour de l'axe Z, vers l'observateur).
//      Unités d'angle : deg | rad | turn | grad
//        1turn  = 360deg  = 2π rad = 400grad

//      transform: rotate(45deg);          /* 1/8 de tour horaire *
//      transform: rotate(-90deg);         /* quart de tour antihoraire *
//      transform: rotate(0.25turn);       /* quart de tour horaire *
//      transform: rotate(1.5708rad);      /* ≈ 90° *

//   4. skew(x, y) / skewX(x) / skewY(y)
//      ────────────────────────────────────
//      Inclinaison / cisaillement (distorsion en parallelogramme).
//      Unités : deg, rad, turn, grad

//      transform: skew(20deg, 10deg);     /* incline sur les 2 axes *
//      transform: skewX(-15deg);          /* incline les colonnes *
//      transform: skewY(10deg);           /* incline les lignes *

//   5. matrix(a, b, c, d, e, f)
//      ──────────────────────────
//      Transformation affine 2D complète via une matrice 3×3 :
//        [ a  c  e ]
//        [ b  d  f ]
//        [ 0  0  1 ]

//      a, d → scale  |  b, c → skew  |  e, f → translate

//   Fonction                 | Dim | Ce qu'elle fait
//   ─────────────────────────|─────|──────────────────────────────────────
//   translate(x, y)          |  2D | Déplace sur X et Y
//   translateX(v)            |  2D | Déplace sur X uniquement
//   translateY(v)            |  2D | Déplace sur Y uniquement
//   translate3d(x, y, z)     |  3D | Déplace dans le volume
//   translateZ(v)            |  3D | Profondeur (rapproche / éloigne)
//   scale(x, y)              |  2D | Agrandit / réduit
//   scaleX(v)                |  2D | Mise à l'échelle horizontale
//   scaleY(v)                |  2D | Mise à l'échelle verticale
//   scale3d(x, y, z)         |  3D | Mise à l'échelle volume
//   scaleZ(v)                |  3D | Étirement en profondeur
//   rotate(a)                |  2D | Rotation plane (= rotateZ)
//   rotateX(a)               |  3D | Bascule avant / arrière
//   rotateY(a)               |  3D | Retournement gauche / droite
//   rotateZ(a)               |  3D | Rotation plane
//   rotate3d(x, y, z, a)     |  3D | Rotation axe quelconque
//   skew(x, y)               |  2D | Cisaillement sur 2 axes
//   skewX(v)                 |  2D | Cisaillement horizontal
//   skewY(v)                 |  2D | Cisaillement vertical
//   perspective(v)           |  3D | Distance observateur (local)
//   matrix(a,b,c,d,e,f)      |  2D | Matrice affine complète 3×3
//   matrix3d(16 val.)        |  3D | Matrice homogène complète 4×4
//   ─────────────────────────|─────|──────────────────────────────────────

//   Propriété                | Rôle
//   ─────────────────────────|──────────────────────────────────────────
//   transform-origin         | Point de pivot de la transformation
//   perspective              | Profondeur 3D (sur le parent)
//   perspective-origin       | Position de l'observateur
//   transform-style          | flat vs preserve-3d pour les enfants
//   backface-visibility      | Cache / montre le verso d'un élément 3D
//   will-change              | Optimisation GPU anticipée
//   ─────────────────────────|──────────────────────────────────────────
