const routes = {
    '/': 'Accueil',         
    '/login': 'Connexion',     
    '/Inscription': 'Inscription'
};

const render = async (path) => {
    const app = document.getElementById('app');
    if (!app) return;

    const pageName = routes[path];

    let promesse = new Promise((resolve, reject) => {
        setTimeout(() => {
            if(pageName){
                resolve("Reussi a recup page")
            }else{
                reject("Page pas reussi")
            }
        }, 3000)
        // arret setTimeout       
    })

    promesse.then(() => {
            console.table("Yes")
        }).catch(() => {
            console.table("error")
        }).finally(() => {
            console.table("FIN")
        })





    let infini = setInterval(() => {
        console.table("je continue")
    }, 3000);
    clearInterval(infini)

    async function getUsers() {
        try {
            const response = await fetch("https://jsonplaceholder.typicode.com/users");
            const data = await response.json();
            console.log(data);
        } catch (error) {
            console.log("Erreur :", error);
        }
    }
    getUsers();

    async function createUser() {
        try {
            const response = await fetch("https://jsonplaceholder.typicode.com/users", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    name: "Amadou",
                    age: 20
                })
            });
            const data = await response.json();
            console.log(data);
        } catch (error) {
            console.log("Erreur :", error);
        }
    }
    createUser();


    

    let pageModule;
    try {
        if (!pageName) {
            pageModule = await import('./pages/NotFound/NotFound.js');
        } else {
            pageModule = await import(`./pages/${pageName}/${pageName}.js`);
        }
        
        const pageComponent = pageModule.default;
        app.innerHTML = pageComponent();

        if (pageComponent.afterRender) {
            pageComponent.afterRender();
        }

    } catch (error) {
        console.error(`Erreur de chargement : ${error}`);
        app.innerHTML = '<h1>Erreur technique</h1>';
    }
};

const navigate = (path) => {
    window.location.hash = path;
};

const handleHashChange = async () => {
    const path = window.location.hash.replace('#', '') || '/';
    await render(path);
};

const initRouter = () => {
    window.addEventListener('hashchange', handleHashChange);
    handleHashChange();
};

export { initRouter, navigate };
